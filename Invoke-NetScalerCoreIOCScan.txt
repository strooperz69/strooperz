<#
.SYNOPSIS
    Read-only IOC scanner for Citrix NetScaler ADC / Gateway core and memory dumps.

.DESCRIPTION
    Scans NetScaler core files, memory dumps, support bundles and collected file trees for
    public indicators tied to:
        CVE-2026-88771 (pitboss log injection RCE)          CTX697096
        CVE-2026-88772 (DTLS pre-auth overflow, WHIPSHOT / SLAPSHOT)  CTX697096
        CVE-2026-88773 (no public IOCs as of the set date)   CTX697096
        CVE-2026-88779 (SAML memory overflow)                CTX697174
    plus the 2025 NetScaler campaign strings published by NCSC-NL.

    How it works
      - Each file is streamed in chunks (default 16 MB) with an overlap window, so large
        cores never load fully into memory and matches across chunk borders are not lost.
      - Bytes are decoded 1:1 (ISO-8859-1), so ASCII strings inside binary data are matched.
      - .gz files (NetScaler cores are usually NSPPE-xx-<pid>.gz) are decompressed on the fly.
        A .tgz support bundle is scanned as one decompressed tar stream; nested .gz cores
        inside it stay compressed, so extract those first.
      - Every regex IOC is also checked against each file path (useful for collected trees).
      - Dotted IPv4 strings are matched against the IP list.
      - SHA-256 of every file is computed and compared with the hash list (skip with -SkipHash).

    Safety
      - Read-only on the evidence. No network access. No code execution of scanned content.
      - Output is written only to -OutputDir. The output folder and this script are excluded
        from the scan so they cannot match themselves.
      - Text taken from evidence is reduced to printable ASCII and guarded against CSV formula
        injection before it is written to disk or the console.
      - Every regex has a match timeout.

    A clean result is not proof of a clean appliance. Atomic IOCs (IPs, paths, hashes) rotate,
    attackers clean up, and some implants live only in memory. Treat hits as leads for triage.

.PARAMETER Path
    One or more files or folders to scan.

.PARAMETER OutputDir
    Folder for results. Default: .\NetScalerIOCScan_<yyyyMMdd_HHmmss> under the current folder.
    Must be new or empty, must not be a drive root, and must not sit inside any -Path, so the
    evidence is never written to. Run the script from a folder outside the evidence.

.PARAMETER Recurse
    Scan folders recursively. Symlinks and junctions found during recursion are skipped and
    logged. A -Path that is itself a link is followed. Other reparse-point files (for example
    deduplicated files) are scanned.

.PARAMETER MinConfidence
    Lowest confidence to report: High, Medium, Low or Hunting (default Hunting = everything).
    Hunting patterns also match legitimate NetScaler data and need analyst review.

.PARAMETER ChunkMB
    Read size per chunk in MB (default 16, max 64).

.PARAMETER MaxHitsPerIoc
    Detailed hit rows kept per IOC per file (default 25). Totals are always counted.

.PARAMETER SkipHash
    Do not compute SHA-256 of scanned files.

.PARAMETER MaxHashMB
    Only hash files up to this size in MB (default 512). Large cores are not hash IOCs and
    hashing them reads the file twice.

.PARAMETER NoGzip
    Scan .gz files as raw bytes instead of decompressing them.

.PARAMETER NoIPs
    Skip the IPv4 indicator check.

.OUTPUTS
    <OutputDir>\hits.csv          one row per hit (capped by -MaxHitsPerIoc)
    <OutputDir>\summary.csv       one row per file with totals, SHA-256 and status
    <OutputDir>\hit_counts.csv    total hits per file per IOC (uncapped)
    <OutputDir>\scan.log          run log
    <OutputDir>\ioc_patterns.csv, ioc_ips.csv, ioc_hashes.csv   the IOC set used

    Exit code (bit flags): 0 = clean, 1 = hits found, 2 = one or more files not fully scanned,
    3 = hits found and some files not fully scanned, 4 = fatal error (run did not complete).

.EXAMPLE
    .\Invoke-NetScalerCoreIOCScan.ps1 -Path 'D:\IR\ns01\var_core' -Recurse

.EXAMPLE
    .\Invoke-NetScalerCoreIOCScan.ps1 -Path 'D:\IR\ns01\NSPPE-00-4511.gz','D:\IR\ns01\nsaaad.core' -MinConfidence Medium

.NOTES
    IOC set date : 2026-10-29
    Requires     : Windows PowerShell 5.1 or PowerShell 7+ (7 is much faster on multi-GB cores;
                   use -MinConfidence High or Medium to cut scan time on 5.1)
    Sources      : Citrix CTX697096 / CTX697174, Google Threat Intelligence (Mandiant), watchTowr
                   (labs blog, intelligence blog, citrix-netscaler-cve-2026-88771-iocs repo),
                   Rapid7, Zscaler ThreatLabz, eSentire, LevelBlue SpiderLabs, Unit 42, CERT-EU,
                   GreyNoise, Lupovis, BleepingComputer, NCSC-NL, pitscaler.com aggregate lists.
    Not included : NCSC-NL 2025 ROP-chain, reversed-string and multi-condition core rules
                   (use NCSC-NL's own tooling for those), Rapid7 Intelligence Hub rules, GTIG
                   registered-only collection, Rubrik Platypus artefacts that are not NetScaler.
#>

function Get-SafeText {
    # Reduce evidence text to printable ASCII, cap length, and neutralise CSV formula prefixes.
    param([string]$Text, [int]$Max = 300)
    if ([string]::IsNullOrEmpty($Text)) { return '' }
    if ($Text.Length -gt $Max) { $Text = $Text.Substring(0, $Max) + '...' }
    $s = [regex]::Replace($Text, '[^\x20-\x7E]', '.')
    if ('=+-@'.IndexOf($s[0]) -ge 0) { $s = "'" + $s }
    return $s
}

function Write-Log {
    param([string]$Message, [string]$Level = 'INFO')
    $safe = Get-SafeText $Message 4000
    $line = '{0} [{1}] {2}' -f (Get-Date -Format 'yyyy-MM-dd HH:mm:ss'), $Level, $safe
    Add-Content -LiteralPath $script:LogFile -Value $line -Encoding UTF8
    if ($Level -eq 'WARN' -or $Level -eq 'ERROR') { Write-Warning $safe } else { Write-Verbose $safe }
}

function Get-FilesToScan {
    # Enumerate files without following reparse points; skip the output folder and this script.
    param([string[]]$Roots, [bool]$Deep, [string]$ExcludeDir, [string]$ExcludeFile)
    $result = New-Object 'System.Collections.Generic.List[string]'
    foreach ($root in $Roots) {
        if (-not (Test-Path -LiteralPath $root)) { Write-Log "Path not found: $root" 'WARN'; continue }
        try { $item = Get-Item -LiteralPath $root -Force -ErrorAction Stop }
        catch { Write-Log ("Cannot open {0}: {1}" -f $root, $_.Exception.Message) 'WARN'; continue }
        if (-not $item.PSIsContainer) { $result.Add($item.FullName); continue }
        if ([System.IO.File]::Exists([System.IO.Path]::Combine($item.FullName, $OutputMarker))) {
            Write-Log ("Skipping prior scan output folder {0}" -f $item.FullName) 'WARN'; continue
        }
        $stack = New-Object 'System.Collections.Generic.Stack[System.IO.DirectoryInfo]'
        $stack.Push([System.IO.DirectoryInfo]$item)
        while ($stack.Count -gt 0) {
            $dir = $stack.Pop()
            try { $entries = $dir.GetFileSystemInfos() }
            catch { Write-Log ("Cannot list {0}: {1}" -f $dir.FullName, $_.Exception.Message) 'WARN'; continue }
            foreach ($e in $entries) {
                $isReparse = (($e.Attributes -band [System.IO.FileAttributes]::ReparsePoint) -ne 0)
                if ($e -is [System.IO.DirectoryInfo]) {
                    if (-not $Deep) { continue }
                    if ($isReparse) { Write-Log ("Skipped linked folder: {0}" -f $e.FullName) 'WARN'; continue }
                    if ($ExcludeDir -and $e.FullName.TrimEnd([char[]]@('\', '/')).Equals($ExcludeDir, [System.StringComparison]::OrdinalIgnoreCase)) { continue }
                    if ([System.IO.File]::Exists([System.IO.Path]::Combine($e.FullName, $OutputMarker))) {
                        Write-Log ("Skipping prior scan output folder {0}" -f $e.FullName) 'WARN'; continue
                    }
                    $stack.Push($e)
                } else {
                    if ($isReparse) {
                        $lt = $null
                        try { $lt = (Get-Item -LiteralPath $e.FullName -Force -ErrorAction Stop).LinkType }
                        catch { Write-Log ("Skipped unreadable reparse point: {0}" -f $e.FullName) 'WARN'; continue }
                        if ($lt -eq 'SymbolicLink' -or $lt -eq 'Junction') {
                            Write-Log ("Skipped linked file: {0}" -f $e.FullName) 'WARN'; continue
                        }
                    }
                    if ($ExcludeFile -and $e.FullName.Equals($ExcludeFile, [System.StringComparison]::OrdinalIgnoreCase)) { continue }
                    if ($ExcludeDir -and $e.FullName.StartsWith($ExcludeDir + [System.IO.Path]::DirectorySeparatorChar, [System.StringComparison]::OrdinalIgnoreCase)) { continue }
                    $result.Add($e.FullName)
                }
            }
        }
    }
    return ,$result
}

function New-HitRow {
    param($File, [int64]$Offset, $Ioc, [string]$Kind, [string]$MatchText, [string]$Snippet)
    [pscustomobject]@{
        File       = Get-SafeText $File 1000
        Offset     = $Offset
        HitType    = $Kind
        IocId      = $Ioc.Id
        CVE        = $Ioc.Cve
        Confidence = $Ioc.Confidence
        Context    = $Ioc.Context
        Source     = $Ioc.Source
        Match      = Get-SafeText $MatchText 300
        Snippet    = Get-SafeText $Snippet 400
    }
}

function Add-Hit {
    # Count every hit; keep detailed rows up to MaxHitsPerIoc per IOC per file.
    # Text is the window text when Index >= 0 (snippet is cut only for kept rows), else the snippet.
    param($State, $Ioc, [string]$Kind, [int64]$Offset, [string]$MatchText, [string]$Text, [int]$Index = -1)
    $key = $Ioc.Id
    if ($State.Counts.ContainsKey($key)) { $State.Counts[$key]++ } else { $State.Counts[$key] = 1 }
    if ($State.Counts[$key] -le $MaxHitsPerIoc) {
        $snip = $Text
        if ($Index -ge 0) { $snip = Get-Snippet $Text $Index $MatchText.Length }
        $State.Rows.Add((New-HitRow $State.File $Offset $Ioc $Kind $MatchText $snip))
    }
}

function Get-Snippet {
    param([string]$Text, [int]$Index, [int]$Length)
    $start = [Math]::Max(0, $Index - 64)
    $len = [Math]::Min($Text.Length - $start, [Math]::Min($Length, 200) + 128)
    return $Text.Substring($start, $len)
}

function Invoke-StreamScan {
    # Window layout per iteration:
    #   buf[0 .. carry)            bytes kept from the previous window (lookbehind margin + overlap)
    #   buf[carry .. filled)       new bytes
    # A match is reported only if it starts in [reportFrom, reportTo). reportTo stops OverlapBytes
    # short of the end (unless at EOF) so a match near the border is reported once, by the next
    # window, with full context on both sides.
    param([System.IO.Stream]$Stream, $State, $Iocs)

    $chunkBytes = $ChunkMB * 1MB
    $carryBytes = $LookbehindMargin + $OverlapBytes
    $buf = New-Object 'byte[]' ($chunkBytes + $carryBytes)
    $carry = 0
    [int64]$windowAbs = 0
    $first = $true

    while ($true) {
        $filled = $carry
        $eof = $false
        while ($filled -lt $buf.Length) {
            $n = $Stream.Read($buf, $filled, $buf.Length - $filled)
            if ($n -le 0) { $eof = $true; break }
            $filled += $n
        }
        if ($filled -le $carry) {
            # No new bytes. If this is not the first window, the tail was deferred and still needs
            # reporting: rescan the carried bytes with reportTo = end.
            if ($first -or $carry -eq 0) { break }
        }
        $State.Bytes = $windowAbs + $filled

        $text = $Latin1.GetString($buf, 0, $filled)
        $reportFrom = if ($first) { 0 } else { $LookbehindMargin }
        $reportTo = if ($eof) { $filled } else { $filled - $OverlapBytes }

        foreach ($ioc in $Iocs) {
            try {
                $m = $ioc.Regex.Match($text, $reportFrom)
                while ($m.Success -and $m.Index -lt $reportTo) {
                    Add-Hit $State $ioc 'Content' ($windowAbs + $m.Index) $m.Value $text $m.Index
                    $m = $m.NextMatch()
                }
            } catch [System.Text.RegularExpressions.RegexMatchTimeoutException] {
                Write-Log ("Regex timeout for {0} in {1} near offset {2}" -f $ioc.Id, $State.File, $windowAbs) 'WARN'
                $State.Status = 'Partial (regex timeout)'
            }
        }

        if (-not $NoIPs -and $null -ne $script:IPv4Regex) {
            try {
                $m = $script:IPv4Regex.Match($text, $reportFrom)
                while ($m.Success -and $m.Index -lt $reportTo) {
                    Add-Hit $State $script:IpIndex[$m.Value] 'IPv4' ($windowAbs + $m.Index) $m.Value $text $m.Index
                    $m = $m.NextMatch()
                }
            } catch [System.Text.RegularExpressions.RegexMatchTimeoutException] {
                Write-Log ("IP regex timeout in {0} near offset {1}" -f $State.File, $windowAbs) 'WARN'
                $State.Status = 'Partial (regex timeout)'
            }
        }

        if ($eof) { break }

        $keep = [Math]::Min($carryBytes, $filled)
        [Array]::Copy($buf, $filled - $keep, $buf, 0, $keep)
        $windowAbs += ($filled - $keep)
        $carry = $keep
        $first = $false

        $pct = 0
        if ($State.Length -gt 0 -and -not $State.Compressed) { $pct = [int][Math]::Min(100, (100.0 * $State.Bytes / $State.Length)) }
        Write-Progress -Id 2 -ParentId 1 -Activity ('Scanning ' + (Get-SafeText ([System.IO.Path]::GetFileName($State.File)) 200)) `
            -Status ('{0:N0} MB' -f ($State.Bytes / 1MB)) -PercentComplete $pct
    }
}

function Test-GzipMagic {
    param([string]$File)
    $fs = [System.IO.File]::Open($File, [System.IO.FileMode]::Open, [System.IO.FileAccess]::Read, [System.IO.FileShare]::ReadWrite)
    try {
        $b = New-Object 'byte[]' 2
        $n = $fs.Read($b, 0, 2)
        return ($n -eq 2 -and $b[0] -eq 0x1F -and $b[1] -eq 0x8B)
    } finally { $fs.Dispose() }
}

        [pscustomobject]@{ File = Get-SafeText $file 1000; SizeBytes = ''; Gzip = ''; BytesScanned = 0; SHA256 = ''; TotalHits = 0; HighRowsKept = 0; DistinctIocs = 0; Status = Get-SafeText ('Not scanned: ' + $_.Exception.Message) 300; Seconds = 0 } |

        File = $file; Length = $fi.Length; Bytes = [int64]0; Compressed = $false; Status = 'OK'

    $pathText = $file.Replace('\', '/')

            if ($m.Success) { Add-Hit $state $ioc 'Path' -1 $m.Value $pathText }
        } catch [System.Text.RegularExpressions.RegexMatchTimeoutException] { }
    }

    # 2. Hash check
    if (-not $SkipHash -and $fi.Length -le ([int64]$MaxHashMB * 1MB)) {
        try {
            $sha = (Get-FileHash -LiteralPath $file -Algorithm SHA256).Hash.ToLowerInvariant()
            if ($HashIndex.ContainsKey($sha)) { Add-Hit $state $HashIndex[$sha] 'SHA256' -1 $sha '' }
        } catch {
            Write-Log ("Hash failed for {0}: {1}" -f $file, $_.Exception.Message) 'WARN'
            $hadErrors = $true
        }
    }

    # 3. Content check
    $fs = $null; $gz = $null
    try {
        $useGzip = (-not $NoGzip) -and $fi.Length -ge 2 -and (Test-GzipMagic $file)
        $fs = [System.IO.File]::Open($file, [System.IO.FileMode]::Open, [System.IO.FileAccess]::Read, [System.IO.FileShare]::ReadWrite)
        if ($useGzip) {
            $state.Compressed = $true
            $gz = New-Object System.IO.Compression.GZipStream($fs, [System.IO.Compression.CompressionMode]::Decompress)
            Invoke-StreamScan -Stream $gz -State $state -Iocs $RegexIocs
            # gzip ISIZE trailer = uncompressed size mod 2^32 of the last member. A mismatch means a
            # truncated core, a multi-member file read only in part, or trailing padding.
            if ($fs.Length -ge 18) {
                $tail = New-Object 'byte[]' 4
                [void]$fs.Seek(-4, [System.IO.SeekOrigin]::End)
                [void]$fs.Read($tail, 0, 4)
                if ([BitConverter]::ToUInt32($tail, 0) -ne [uint32]($state.Bytes % 4294967296)) {
                    $state.Status = 'Partial (gzip size mismatch: truncated, multi-member or padded)'
                    Write-Log ("Gzip size check failed for {0}; content may be incomplete." -f $file) 'WARN'
                }
            }
        } else {
            Invoke-StreamScan -Stream $fs -State $state -Iocs $RegexIocs
        }
    } catch {
        $hadErrors = $true
        $state.Status = 'Partial: ' + $_.Exception.Message
        Write-Log ("Scan incomplete for {0} after {1:N0} bytes: {2}" -f $file, $state.Bytes, $_.Exception.Message) 'WARN'
    } finally {
        if ($gz) { $gz.Dispose() }
        if ($fs) { $fs.Dispose() }
    }
    if ($state.Status -ne 'OK') { $hadErrors = $true }

    $fileHits = 0
    foreach ($v in $state.Counts.Values) { $fileHits += $v }
    $totalHits += $fileHits

    if ($state.Rows.Count -gt 0) {
        $state.Rows | Sort-Object @{ Expression = { $ConfidenceRank[$_.Confidence] }; Descending = $true }, IocId, Offset |
            Export-Csv -LiteralPath $HitsCsv -NoTypeInformation -Encoding UTF8 -Append

        } | Sort-Object IocId | Export-Csv -LiteralPath $CountsCsv -NoTypeInformation -Encoding UTF8 -Append
    }

    $hiCount = @($state.Rows | Where-Object { $_.Confidence -eq 'High' }).Count
    [pscustomobject]@{
        File            = Get-SafeText $file 1000
        SizeBytes       = $fi.Length
        Gzip            = $state.Compressed
        BytesScanned    = $state.Bytes
        SHA256          = $sha
        TotalHits       = $fileHits
        HighRowsKept    = $hiCount
        DistinctIocs    = $state.Counts.Count
        Status          = Get-SafeText $state.Status 300
        Seconds         = [int]((Get-Date) - $started).TotalSeconds
    } | Export-Csv -LiteralPath $SummaryCsv -NoTypeInformation -Encoding UTF8 -Append

    Write-Log ("{0}: {1} hits across {2} IOCs, {3:N0} bytes scanned, status {4}" -f $file, $fileHits, $state.Counts.Count, $state.Bytes, $state.Status)

    foreach ($o in $RegexIocs) { $meta[$o.Id] = $o }
    foreach ($o in $script:IpIndex.Values) { $meta[$o.Id] = $o }
    foreach ($o in $HashIndex.Values) { $meta[$o.Id] = $o }
    $counts | ForEach-Object {
        $m = $meta[$_.IocId]
        [pscustomobject]@{
            Confidence = if ($m) { $m.Confidence } else { '' }
            IocId      = $_.IocId
            Hits       = [int]$_.Hits
            File       = Split-Path -Leaf $_.File
            Context    = if ($m) { $m.Context } else { '' }
        }
    } | Sort-Object @{ Expression = { $ConfidenceRank[$_.Confidence] }; Descending = $true }, File, IocId |
        Format-Table -AutoSize -Wrap | Out-String -Width 220 | Write-Host
}
if ($hadErrors) { Write-Host 'One or more files were not fully scanned. See scan.log.' -ForegroundColor Yellow }
Write-Log ("Scan finished. Total hits {0}. Errors {1}." -f $totalHits, $hadErrors)

$code = 0
if ($totalHits -gt 0) { $code = $code + 1 }
if ($hadErrors) { $code = $code + 2 }
exit $code
